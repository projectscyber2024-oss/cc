
#include <stdio.h>
#include <string.h>

int main()
{
    char text[] = "Hello";
    char key[] = "KEY";

    int S[256];
    int i, j = 0, temp;
    int keylen = strlen(key);
    int len = strlen(text);

    unsigned char cipher[100];

    for (i = 0; i < 256; i++)
        S[i] = i;

    for (i = 0; i < 256; i++)
    {
        j = (j + S[i] + key[i % keylen]) % 256;

        temp = S[i];
        S[i] = S[j];
        S[j] = temp;
    }

    i = 0;
    j = 0;

    for (int k = 0; k < len; k++)
    {
        i = (i + 1) % 256;
        j = (j + S[i]) % 256;

        temp = S[i];
        S[i] = S[j];
        S[j] = temp;

        int t = (S[i] + S[j]) % 256;

        cipher[k] = text[k] ^ S[t];
    }

    printf("Encrypted text: ");

    for (i = 0; i < len; i++)
        printf("%02X ", cipher[i]);

    printf("\n");

    return 0;
}
